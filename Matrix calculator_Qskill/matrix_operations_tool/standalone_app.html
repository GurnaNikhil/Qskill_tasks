import sys
import numpy as np

def get_positive_integer(prompt):
    while True:
        try:
            val = int(input(prompt).strip())
            if val <= 0:
                print("Error: Value must be a positive integer greater than 0.")
                continue
            return val
        except ValueError:
            print("Error: Invalid input. Please enter a valid integer.")

def input_matrix(name="Matrix"):
    print(f"\n--- Enter Details for {name} ---")
    rows = get_positive_integer(f"Enter number of rows for {name}: ")
    cols = get_positive_integer(f"Enter number of columns for {name}: ")
    print(f"\nEnter the elements for {name} ({rows}x{cols}):")
    matrix_rows = []
    i = 0
    while i < rows:
        row_input = input(f"Row {i + 1} ({cols} values): ").strip()
        parts = row_input.split()
        if len(parts) != cols:
            print(f"Error: Expected exactly {cols} values.")
            continue
        try:
            row_values = [float(val) if '.' in val else int(val) for val in parts]
            matrix_rows.append(row_values)
            i += 1
        except ValueError:
            print("Error: Non-numeric element detected.")
    return np.array(matrix_rows)

def matrix_addition():
    a = input_matrix("Matrix A")
    b = input_matrix("Matrix B")
    if a.shape != b.shape:
        print("Error: Matrix dimensions must be the same for addition.")
        return
    res = np.add(a, b)
    print("Result (A + B):\n", res)

def matrix_subtraction():
    a = input_matrix("Matrix A")
    b = input_matrix("Matrix B")
    if a.shape != b.shape:
        print("Error: Matrix dimensions must be the same for subtraction.")
        return
    res = np.subtract(a, b)
    print("Result (A - B):\n", res)

def matrix_multiplication():
    a = input_matrix("Matrix A")
    b = input_matrix("Matrix B")
    if a.shape[1] != b.shape[0]:
        print("Error: Matrix A columns must equal Matrix B rows for multiplication.")
        return
    res = np.matmul(a, b)
    print("Result (A x B):\n", res)

def matrix_transpose():
    a = input_matrix("Matrix A")
    print("Transposed Matrix A^T:\n", np.transpose(a))

def matrix_determinant():
    a = input_matrix("Matrix A")
    if a.shape[0] != a.shape[1]:
        print("Error: Determinant can only be calculated for a square matrix.")
        return
    print("Determinant |A|:", round(float(np.linalg.det(a)), 4))

def main():
    while True:
        print("\n1. Addition\n2. Subtraction\n3. Multiplication\n4. Transpose\n5. Determinant\n6. Exit")
        c = input("Enter choice: ").strip()
        if c == "1": matrix_addition()
        elif c == "2": matrix_subtraction()
        elif c == "3": matrix_multiplication()
        elif c == "4": matrix_transpose()
        elif c == "5": matrix_determinant()
        elif c == "6": break

if __name__ == "__main__":
    main()
